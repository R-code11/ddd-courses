Question 3 (b) function defining to calculate mean deviation

mean_dev <- function(x) {
  x - mean(x, na.rm = TRUE)}
mean_dev(df$a)
[1]  0.2112923  0.5013320  1.0093820 -0.9733525 -0.4724039 -0.2762499
mean_dev(df$b)
[1] -1.0451072 -1.0951159  1.1179410  0.2643910 -0.1841368  0.9420279
## The output of the above code will be the mean deviation of each value in columns a and b from their means. The mean deviation is calculated by subtracting the mean of the column from each individual value in that column. The output will be displayed here;

# STEP 1: Create family size and constructed per capita income
FINAL_EXAM$HH_DATA_q1 <- HH_DATA |>
  mutate(

  ) |>
  select(INCOME, family_size, income_pc_constructed, INCOMEPC)

# Check for missing values
summary(FINAL_EXAM$HH_DATA_q1)

FINAL_EXAM$df_q1_clean <- FINAL_EXAM$HH_DATA_q1 |>
  filter(!is.na(income_pc_constructed), 
         !is.na(INCOMEPC),
         income_pc_constructed > 0,
         INCOMEPC > 0)

# STEP 3: Plot correlation between constructed and given per capita income
FINAL_EXAM$df_q1_clean |>
  ggplot(aes(x = INCOMEPC, y = income_pc_constructed)) +
  geom_point(alpha = 0.5, size = 2) +
  geom_smooth(se = TRUE, method = "lm", color = "red") +
  labs(
    title = "Correlation between Constructed and Given Per Capita Income (Ramesh Beniwal)",
    x = "Given Per Capita Income (Rupees) [Ramesh Beniwal]",
    y = "Constructed Per Capita Income (Rupees) [Ramesh Beniwal]"
  ) +
  theme_minimal() +
  theme(
    plot.title = element_text(hjust = 0.5, face = "bold", size = 12),
    axis.title = element_text(size = 10)
  )

# STEP 4: Calculate correlation coefficient
FINAL_EXAM$correlation_q1 <- cor(FINAL_EXAM$df_q1_clean$INCOMEPC,

# QUESTION 2: Validate Income-Consumption Assumption
# =====================================================

# STEP 1: Prepare data with both income variables
FINAL_EXAM$HH_DATA_q2 <- HH_DATA |>
  mutate(
    family_size = NADULTM + NADULTF + NCHILDM + NCHILDF + NTEENM + NTEENF + NELDERM + NELDERF,
    income_pc_constructed = INCOME / family_size
  ) |>
  select(income_pc_constructed, INCOMEPC, COPC) |>
  filter(income_pc_constructed > 0, INCOMEPC > 0, COPC > 0)

# Check data
print("Summary before outlier removal (Q2):")
summary(FINAL_EXAM$HH_DATA_q2)

# STEP 2: Remove outliers (filter high values like you did in scripts)
FINAL_EXAM$HH_DATA_q2 <- FINAL_EXAM$HH_DATA_q2 |>
  filter(COPC < 100000, income_pc_constructed < 100000, INCOMEPC < 100000)

# Check cleaned data
print("Summary after outlier removal (Q2):")
summary(FINAL_EXAM$HH_DATA_q2)

# STEP 3: PLOT 1 - Both relationships on same graph
FINAL_EXAM$HH_DATA_q2 |>
  ggplot() +
  geom_smooth(aes(x = INCOMEPC, y = COPC, color = "Given Per Capita Income"), 
              se = TRUE, method = "lm", linewidth = 1.2) +
  geom_smooth(aes(x = income_pc_constructed, y = COPC, color = "Constructed Per Capita Income"), 
              se = TRUE, method = "lm", linewidth = 1.2) +
  scale_x_continuous(labels = scales::comma) +
  scale_y_continuous(labels = scales::comma) +
  labs(
    title = "Income-Consumption Relationship: Validation of Assumption (Ramesh Beniwal)",
    x = "Per Capita Income (Rupees) [Ramesh Beniwal]",
    y = "Per Capita Consumption (Rupees) [Ramesh Beniwal]",
    color = "Income Measure [Ramesh Beniwal]"
  ) +
  theme_minimal() +
  theme(
    plot.title = element_text(hjust = 0.5, face = "bold", size = 12),
    axis.title = element_text(size = 10),
    legend.position = "bottom"
  )

# STEP 4: PLOT 2 - Faceted comparison
FINAL_EXAM$HH_DATA_q2_long <- FINAL_EXAM$HH_DATA_q2 |>
  pivot_longer(cols = c(INCOMEPC, income_pc_constructed),
               names_to = "income_type",
               values_to = "income_value") |>
  mutate(income_type = case_when(
    income_type == "INCOMEPC" ~ "Given Per Capita Income",
    income_type == "income_pc_constructed" ~ "Constructed Per Capita Income"
  ))

FINAL_EXAM$HH_DATA_q2_long |>
  ggplot(aes(x = income_value, y = COPC, color = income_type)) +
  geom_point(alpha = 0.3, size = 1.5) +
  geom_smooth(se = TRUE, method = "lm", linewidth = 1) +
  facet_wrap(~income_type, scales = "free_x") +
  scale_x_continuous(labels = scales::comma) +
  scale_y_continuous(labels = scales::comma) +
  labs(
    title = "Income-Consumption Relationship by Income Type (Ramesh Beniwal)",
    x = "Per Capita Income (Rupees) [Ramesh Beniwal]",
    y = "Per Capita Consumption (Rupees) [Ramesh Beniwal]",
    color = "Income Measure [Ramesh Beniwal]"
  ) +
  theme_minimal() +
  theme(
    plot.title = element_text(hjust = 0.5, face = "bold", size = 12),
    axis.title = element_text(size = 10),
    legend.position = "bottom"
  )

# STEP 5: Calculate correlations
FINAL_EXAM$cor_given_income <- cor(FINAL_EXAM$HH_DATA_q2$INCOMEPC, FINAL_EXAM$HH_DATA_q2$COPC)
FINAL_EXAM$cor_constructed_income <- cor(FINAL_EXAM$HH_DATA_q2$income_pc_constructed, FINAL_EXAM$HH_DATA_q2$COPC)

# STEP 6: Print results
print("QUESTION 2: VALIDATION OF INCOME-CONSUMPTION ASSUMPTION")
