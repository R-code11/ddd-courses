# -----------------------------------------------
# Part (b): Write Z-score function
# Z-score formula: (value - mean) / standard deviation
# na.rm = TRUE means ignore NA values during calculation
# -----------------------------------------------

MIDSEM_DDD$z_score <- function(x) {
  (x - mean(x, na.rm = TRUE)) / sd(x, na.rm = TRUE)
} # x is the input numeric vector
  # mean(x, na.rm=TRUE) calculates average, skipping NA values (same for sd also)
 

# -----------------------------------------------
# Part (c): Apply function to transform the data frame
# mutate() creates or changes columns
# -----------------------------------------------

MIDSEM_DDD$df3_change <- MIDSEM_DDD$df3 %>%
  mutate(
    # Replace columns a, b, c with their z-scores
    across(c(a, b, c), MIDSEM_DDD$z_score),

    # New variable mean_deviation = each value of d minus the mean of d
    mean_deviation = d - mean(d, na.rm = TRUE),

    # New variable sum_z = sum of z-scores of a, b, c for each row
    # rowSums() adds values across columns row by row
    sum_z = rowSums(across(c(a, b, c)), na.rm = TRUE)
  )

# Step 1: Create a numeric vector 
MIDSEM_DDD$vec <- c(4, 8, 15, 16, 23, 42)

# Step 2: Calculate the mean of the full vector ONCE before the loop
MIDSEM_DDD$vec_mean <- mean(MIDSEM_DDD$vec)   # calculates the average

# Step 3: Write the absolute deviation function
MIDSEM_DDD$abs_dev <- function(x, mean_val) {
  sqrt((x - mean_val)^2)
}

# Step 4: Preallocate empty output vector 
# numeric(length(vec)) creates a vector of zeros, same size as vec
MIDSEM_DDD$result4 <- numeric(length(MIDSEM_DDD$vec))

# Step 5: Apply function element by element using for loop
# i goes from 1 to length of vec
for (i in 1:length(MIDSEM_DDD$vec)) {
  MIDSEM_DDD$result4[i] <- MIDSEM_DDD$abs_dev(MIDSEM_DDD$vec[i], MIDSEM_DDD$vec_mean)
}
print(MIDSEM_DDD$result4)
[1] 14 10  3  2  5 24
# -----------------------------------------------
# WITHOUT for loop - same result in one line
# -----------------------------------------------
# sqrt() of squared difference gives absolute deviation
MIDSEM_DDD$result4_no_loop <- sqrt((MIDSEM_DDD$vec - mean(MIDSEM_DDD$vec))^2)
print(MIDSEM_DDD$result4_no_loop)
[1] 14 10  3  2  5 24

Question 7

# Step 1: Create a vector of marks for 10 students
MIDSEM_DDD$marks <- c(95, 82, 67, 55, 78, 91, 60, 45, 73, 88)

# Step 2: Preallocate empty character vector to store grades
MIDSEM_DDD$grades <- character(length(MIDSEM_DDD$marks))

# Step 3: For loop with if-else to check each student's marks
for (i in 1:length(MIDSEM_DDD$marks)) {

  if (MIDSEM_DDD$marks[i] >= 90) {   # marks >= 90 gets Grade A
    MIDSEM_DDD$grades[i] <- "Grade A"

  } else if (MIDSEM_DDD$marks[i] >= 75) { # marks >= 75 but less than 90 gets Grade B
    MIDSEM_DDD$grades[i] <- "Grade B"

  } else if (MIDSEM_DDD$marks[i] >= 60) {    # marks >= 60 but less than 75 gets Grade C
    MIDSEM_DDD$grades[i] <- "Grade C"

  } else {      # marks below 60 gets Fail
    MIDSEM_DDD$grades[i] <- "Fail"
  }
}


print(MIDSEM_DDD$grades)
 [1] "Grade A" "Grade B" "Grade C" "Fail"    "Grade B" "Grade A" "Grade C"
 [8] "Fail"    "Grade C" "Grade B"
