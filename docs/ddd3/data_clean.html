Women’s data: recoding

W_SLUMS <- W_SLUMS %>%
  mutate(
    # widow or separated = 1, others = 0
    widow_sep = ifelse(marital_status %in% c(3, 4), 1, 0),
    # age groups
    age_group = case_when(
      age < 18               ~ 1,    # below 18
      age >= 18 & age < 25   ~ 2,    # 18-24
      age >= 25 & age < 35   ~ 3,    # 25-34
      age >= 35 & age < 45   ~ 4,    # 35-44
      age >= 45 & age < 55   ~ 5,    # 45-54
      age >= 55 & age < 65   ~ 6,    # 55-64
      age >= 65              ~ 7     # 65+
    ),
    
    # factors
    slum_id      = as.factor(slum_id),
    joint_family = as.integer(joint_family == 1),
    fil_alive    = as.integer(fil_alive == 1),
    mil_alive    = as.integer(mil_alive == 1))

W_SLUMS <- W_SLUMS %>%
  mutate(
    edu_cat = case_when(
      education == 0                    ~ 1,   # Illiterate
      education >= 1  & education <= 5  ~ 2,   # Primary
      education >= 6  & education <= 8  ~ 3,   # Middle
      education >= 9  & education <= 10 ~ 4,   # Secondary
      education >= 11 & education <= 12 ~ 5,   # Higher Secondary
      education >= 13 & education <= 16 ~ 6    # Graduate & Above
    ))

Widowed or separated indicator, age groups and education categories.

    edu_secondary    = as.integer(edu_cat == 4),
    edu_higher_sec   = as.integer(edu_cat == 5),
    edu_graduate     = as.integer(edu_cat == 6),
    
    # age group dummies
    age_below_18     = as.integer(age_group == 1),
    age_18_24        = as.integer(age_group == 2),

    age_35_44        = as.integer(age_group == 4),
    age_45_54        = as.integer(age_group == 5),

  )
